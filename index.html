async function loadUrl() { let inputUrl = document.getElementById('addressBar').value.trim(); const viewPane = document.getElementById('viewPane'); if (!inputUrl) return; if (!/^https?:\/\//i.test(inputUrl)) { inputUrl = 'https://' + inputUrl; document.getElementById('addressBar').value = inputUrl; } viewPane.innerHTML = `

Proxying request via secure tunnel...

`; // Primary Proxy (corsproxy.io works excellently on GitHub Pages) const primaryProxy = `https://corsproxy.io{encodeURIComponent(inputUrl)}`; // Backup Proxy (allorigins) const backupProxy = `https://allorigins.win{encodeURIComponent(inputUrl)}`; try { // Try Primary Proxy First let response = await fetch(primaryProxy); if (!response.ok) { console.log("Primary proxy failed, trying fallback..."); // Try Backup Proxy if the first one fails response = await fetch(backupProxy); if (!response.ok) throw new Error(`Both proxy nodes failed (${response.status})`); const data = await response.json(); renderHtml(data.contents, inputUrl); } else { const htmlText = await response.text(); renderHtml(htmlText, inputUrl); } } catch (error) { console.error("Detailed Network Error:", error); viewPane.innerHTML = `

Proxy Connection Refused

Error Details: ${error.message}

Why this happens on GitHub Pages:

  • The target site might have strict security rules (CSPs) that prevent it from being loaded inside another webpage.
  • An ad-blocker or privacy extension in your browser might be blocking the public proxy URL. Try turning off your ad-blocker for your GitHub website.
`; } } function renderHtml(rawHtml, originalUrl) { const viewPane = document.getElementById('viewPane'); const parser = new DOMParser(); const doc = parser.parseFromString(rawHtml, 'text/html'); const baseUrl = new URL(originalUrl).origin; // Fix absolute resource routing doc.querySelectorAll('[src]').forEach(el => { const src = el.getAttribute('src'); if (src && !src.startsWith('http') && !src.startsWith('//')) { el.setAttribute('src', `${baseUrl}/${src.replace(/^\//, '')}`); } }); doc.querySelectorAll('[href]').forEach(el => { const href = el.getAttribute('href'); if (href && !href.startsWith('http') && !href.startsWith('//')) { el.setAttribute('href', `${baseUrl}/${href.replace(/^\//, '')}`); } }); viewPane.innerHTML = doc.documentElement.innerHTML; }
